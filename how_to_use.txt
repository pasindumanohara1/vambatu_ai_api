## 2. JavaScript (Browser / Node)
```html
<script>
const uid = self.crypto.randomUUID(); // unique per tab
async function chat(msg) {
  const res = await fetch("https://vambatu-ai-api.onrender.com/chat", {
    method: "POST",
    headers: {"Content-Type": "application/json"},
    body: JSON.stringify({uid, role: "user", text: msg})
  });
  const data = await res.json();
  return data.reply; // Sinhala reply
}

// use
chat("මට joke එකක් කියන්න").then(r => console.log("වඹටු:", r));
</script>
```

## 3. React Hook
```js
import { useState } from "react";

export default function useVambatu() {
  const [reply, setReply] = useState("");
  const uid = self.crypto.randomUUID();

  const send = async (text) => {
    const res = await fetch("https://vambatu-ai-api.onrender.com/chat", {
      method: "POST",
      headers: {"Content-Type": "application/json"},
      body: JSON.stringify({uid, role: "user", text})
    });
    const data = await res.json();
    setReply(data.reply);
  };

  return { reply, send };
}
```
Usage:
```jsx
const { reply, send } = useVambatu();
send("Galle Face යමුද?");
return <p>{reply}</p>;
```

## 4. Python (any version)
```python
import requests, json

URL = "https://vambatu-ai-api.onrender.com/chat"

def vambatu_chat(text: str, uid: str) -> str:
    r = requests.post(URL, json={"uid": uid, "role": "user", "text": text}, timeout=15)
    return r.json()["reply"]

# use
print(vambatu_chat("මට trip plan එකක් දෙන්න", "py-demo"))
```

## 5. Flutter / Dart
```dart
import 'package:http/http.dart' as http;

Future<String> vambatuChat(String text, String uid) async {
  final res = await http.post(
    Uri.parse("https://vambatu-ai-api.onrender.com/chat"),
    headers: {"Content-Type": "application/json"},
    body: jsonEncode({"uid": uid, "role": "user", "text": text}),
  );
  return jsonDecode(res.body)["reply"];
}

// use
final reply = await vambatuChat("හෙලෝ", "flutter-demo");
print(reply); // සිංහල reply
```
